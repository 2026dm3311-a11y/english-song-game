let teams = [];

let currentTeam = 0;

let boardData = [];

let selectedIndex = -1;

let answered = false;



/* =====================================================
   팀 추가
===================================================== */

function addTeam() {

  const input =
    document.getElementById("teamInput");

  const name =
    input.value.trim();


  if(name === "") {

    alert("팀 이름을 입력하세요!");

    return;

  }


  if(teams.length >= 8) {

    alert("최대 8팀까지 가능합니다!");

    return;

  }


  teams.push({

    name: name,

    score: 0

  });


  input.value = "";

  showTeams();

}



/* =====================================================
   팀 표시
===================================================== */

function showTeams() {

  const list =
    document.getElementById("teamList");

  list.innerHTML = "";


  teams.forEach(function(team, index) {

    const box =
      document.createElement("div");

    box.className = "teamBox";

    box.textContent =
      "👥 " +
      (index + 1) +
      "팀 : " +
      team.name;

    list.appendChild(box);

  });

}



/* =====================================================
   게임 시작

function startGame() {

  if(teams.length < 2) {

    alert("최소 2팀을 추가하세요!");

    return;

  }


  document.getElementById("setup")
    .style.display = "none";


  document.getElementById("game")
    .style.display = "block";


  createBoard();

  updateTurn();

  updateScores();

}



/* =====================================================
   보드 생성

   35곡 + 꽝 5개를 하나의 배열에 넣고
   배열 전체를 랜덤으로 섞는다.

   ★ 번호와 정답이 절대 따로 섞이지 않음 ★
===================================================== */

function createBoard() {

  const board =
    document.getElementById("board");

  board.innerHTML = "";

  boardData = [];


  /* 35곡 추가 */

  songs.forEach(function(song) {

    boardData.push({

      type: "song",

      title: song[0],

      artist: song[1],

      used: false

    });

  });


  /* 꽝 5개 추가 */

  for(let i = 0; i < 5; i++) {

    boardData.push({

      type: "boom",

      title: null,

      artist: null,

      used: false

    });

  }


  /* =================================================
     40개 전체 랜덤 섞기
  ================================================= */

  for(let i = boardData.length - 1; i > 0; i--) {

    const randomIndex =
      Math.floor(Math.random() * (i + 1));


    const temp =
      boardData[i];


    boardData[i] =
      boardData[randomIndex];


    boardData[randomIndex] =
      temp;

  }


  /* 40개 번호 만들기 */

  for(let i = 0; i < 40; i++) {

    const cell =
      document.createElement("div");

    cell.className = "cell";

    cell.textContent = i + 1;


    cell.onclick = function() {

      selectCell(i);

    };


    board.appendChild(cell);

  }

}



/* =====================================================
   번호 선택
===================================================== */

function selectCell(index) {

  if(answered) {

    return;

  }


  if(boardData[index].used) {

    return;

  }


  selectedIndex = index;


  const data =
    boardData[index];


  data.used = true;


  const cells =
    document.querySelectorAll(".cell");


  cells[index].classList.add("used");


  document.getElementById("questionArea")
    .style.display = "block";


  document.getElementById("numberText")
    .textContent =
      "🔢 " + (index + 1) + "번";


  document.getElementById("result")
    .textContent = "";


  document.getElementById("nextButton")
    .style.display = "none";


  /* 꽝인 경우 */

  if(data.type === "boom") {

    document.getElementById("youtubeButton")
      .style.display = "none";


    document.getElementById("normalArea")
      .style.display = "none";


    document.getElementById("boomArea")
      .innerHTML =

      "<p style='font-size:40px;font-weight:bold;'>" +
      "💥 꽝!" +
      "</p>" +

      "<button class='boom' onclick='boomAnswer()'>" +
      "💥 꽝 처리 (-1점)" +
      "</button>";


    return;

  }


  /* 노래인 경우 */

  document.getElementById("youtubeButton")
    .style.display = "inline-block";


  document.getElementById("normalArea")
    .style.display = "block";


  document.getElementById("boomArea")
    .innerHTML = "";


  document.getElementById("answerInput")
    .value = "";


  document.getElementById("answerInput")
    .focus();

}



/* =====================================================
   유튜브 연결

   선택된 번호의 정확한 노래 제목 + 가수로
   유튜브 검색을 열어준다.
===================================================== */

function openYouTube() {

  if(selectedIndex === -1) {

    return;

  }


  const data =
    boardData[selectedIndex];


  if(data.type !== "song") {

    return;

  }


  const query =
    data.title +
    " " +
    data.artist +
    " official music video";


  const url =
    "https://www.youtube.com/results?search_query=" +
    encodeURIComponent(query);


  window.open(url, "_blank");

}



/* =====================================================
   정답 확인
===================================================== */

function checkAnswer() {

  if(answered) {

    return;

  }


  if(selectedIndex === -1) {

    return;

  }


  const data =
    boardData[selectedIndex];


  if(data.type !== "song") {

    return;

  }


  const input =
    document.getElementById("answerInput")
      .value;


  if(input.trim() === "") {

    alert("노래 제목을 입력하세요!");

    return;

  }


  /* =================================================
     ★ 핵심 정답 판정 ★

     대문자/소문자 무시
     띄어쓰기 무시
  ================================================= */

  const userAnswer =
    normalize(input);


  const correctAnswer =
    normalize(data.title);


  if(userAnswer === correctAnswer) {

    /* 정답 +2 */

    teams[currentTeam].score += 2;

    answered = true;


    showEffect("🎉 +2점");


    document.getElementById("result")
      .innerHTML =
      "🎉 정답! <strong>+2점</strong>";

  }

  else {

    /* 오답 0 */

    answered = true;


    showEffect("❌ 0점");


    document.getElementById("result")
      .innerHTML =
      "❌ 오답! <strong>0점</strong>";

  }


  updateScores();


  document.getElementById("nextButton")
    .style.display = "inline-block";

}



/* =====================================================
   정답 비교 함수

   ★ 대문자/소문자 무시
   ★ 띄어쓰기 무시

   예:
   Shape of You
   shapeofyou
   SHAPE OF YOU

   모두 같은 것으로 인식
===================================================== */

function normalize(text) {

  return text

    .toLowerCase()

    .replace(/\s+/g, "")

    .trim();

}



/* =====================================================
   꽝 처리
===================================================== */

function boomAnswer() {

  if(answered) {

    return;

  }


  teams[currentTeam].score -= 1;


  answered = true;


  showEffect("💥 -1점");


  document.getElementById("result")
    .innerHTML =
    "💥 꽝! <strong>-1점</strong>";


  document.getElementById("nextButton")
    .style.display = "inline-block";


  updateScores();

}



/* =====================================================
   다음 팀
===================================================== */

function nextTurn() {

  document.getElementById("questionArea")
    .style.display = "none";


  document.getElementById("answerInput")
    .value = "";


  document.getElementById("boomArea")
    .innerHTML = "";


  document.getElementById("result")
    .textContent = "";


  selectedIndex = -1;

  answered = false;


  /* 자동 팀 순서 */

  currentTeam++;


  if(currentTeam >= teams.length) {

    currentTeam = 0;

  }


  updateTurn();

}



/* =====================================================
   현재 팀 표시
===================================================== */

function updateTurn() {

  document.getElementById("turn")
    .textContent =
      "🎯 현재 차례 : " +
      teams[currentTeam].name +
      "팀";

}



/* =====================================================
   점수판
===================================================== */

function updateScores() {

  const scores =
    document.getElementById("scores");


  scores.innerHTML = "";


  teams.forEach(function(team, index) {

    const row =
      document.createElement("div");


    row.className = "scoreRow";


    row.innerHTML =

      "<span>" +
      "👥 " +
      team.name +
      "</span>" +

      "<strong>" +
      team.score +
      "점" +
      "</strong>";


    scores.appendChild(row);

  });

}



/* =====================================================
   큰 점수 효과
===================================================== */

function showEffect(text) {

  const effect =
    document.getElementById("effect");


  effect.textContent = text;


  effect.style.display = "flex";


  setTimeout(function() {

    effect.style.display = "none";

  }, 900);

}
